package client

// Probe (temporary test, not part of any change): what the general
// estimator (cluster ResourceSummary, no resource models) answers.

import (
	"context"
	"testing"

	corev1 "k8s.io/api/core/v1"
	"k8s.io/apimachinery/pkg/api/resource"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"

	clusterv1alpha1 "github.com/karmada-io/karmada/pkg/apis/cluster/v1alpha1"
	workv1alpha2 "github.com/karmada-io/karmada/pkg/apis/work/v1alpha2"
)

const k014GPU = corev1.ResourceName("nvidia.com/gpu")

func k014Cluster(name string, nodes, gpusPerNode int64) *clusterv1alpha1.Cluster {
	return &clusterv1alpha1.Cluster{
		ObjectMeta: metav1.ObjectMeta{Name: name},
		Status: clusterv1alpha1.ClusterStatus{ResourceSummary: &clusterv1alpha1.ResourceSummary{
			Allocatable: corev1.ResourceList{
				corev1.ResourceCPU:    *resource.NewQuantity(64*nodes, resource.DecimalSI),
				corev1.ResourceMemory: *resource.NewQuantity(512*1024*1024*1024*nodes, resource.BinarySI),
				corev1.ResourcePods:   *resource.NewQuantity(110*nodes, resource.DecimalSI),
				k014GPU:               *resource.NewQuantity(nodes*gpusPerNode, resource.DecimalSI),
			},
		}},
	}
}

func k014Req(cpu string, mem string, gpus int64) corev1.ResourceList {
	return corev1.ResourceList{
		corev1.ResourceCPU:    resource.MustParse(cpu),
		corev1.ResourceMemory: resource.MustParse(mem),
		k014GPU:               *resource.NewQuantity(gpus, resource.DecimalSI),
	}
}

func TestK014ProbeGeneralEstimator(t *testing.T) {
	clusters := []*clusterv1alpha1.Cluster{k014Cluster("frag(2 nodes x 2 GPU)", 2, 2), k014Cluster("big(1 node x 8 GPU)", 1, 8)}
	ge := NewGeneralEstimator()
	type shape struct {
		name       string
		sum        corev1.ResourceList
		maxPod     corev1.ResourceList
		components []workv1alpha2.Component
	}
	comp := func(name string, replicas int32, gpus int64) workv1alpha2.Component {
		return workv1alpha2.Component{Name: name, Replicas: replicas, ReplicaRequirements: &workv1alpha2.ComponentReplicaRequirements{ResourceRequest: k014Req("1", "1Gi", gpus)}}
	}
	shapes := []shape{
		{"4x1GPU(entry1+3 workers)", k014Req("4", "4Gi", 4), k014Req("1", "1Gi", 1), []workv1alpha2.Component{comp("entry", 1, 1), comp("worker", 3, 1)}},
		{"3GPU+1GPU(entry+1 worker)", k014Req("2", "2Gi", 4), k014Req("1", "1Gi", 3), []workv1alpha2.Component{comp("entry", 1, 3), comp("worker", 1, 1)}},
	}
	for _, s := range shapes {
		sum, err1 := ge.MaxAvailableReplicas(context.TODO(), ReplicaEstimationRequest{Clusters: clusters, ReplicaRequirements: &workv1alpha2.ReplicaRequirements{ResourceRequest: s.sum}})
		maxPod, err2 := ge.MaxAvailableReplicas(context.TODO(), ReplicaEstimationRequest{Clusters: clusters, ReplicaRequirements: &workv1alpha2.ReplicaRequirements{ResourceRequest: s.maxPod}})
		sets, err3 := ge.MaxAvailableComponentSets(context.TODO(), ComponentSetEstimationRequest{Clusters: clusters, Components: s.components, Namespace: "default"})
		t.Logf("K014 general | %s | replicas(sum)=%v err=%v | replicas(maxPod)=%v err=%v | componentSets=%v err=%v", s.name, sum, err1, maxPod, err2, sets, err3)
	}
}
