package status

// Probe (temporary test, not part of any change): with Karmada's real default
// cluster resource models, what does the general estimator answer for a GPU request?

import (
	"context"
	"testing"

	corev1 "k8s.io/api/core/v1"
	"k8s.io/apimachinery/pkg/api/resource"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"

	clusterapis "github.com/karmada-io/karmada/pkg/apis/cluster"
	"github.com/karmada-io/karmada/pkg/apis/cluster/mutation"
	clusterv1alpha1 "github.com/karmada-io/karmada/pkg/apis/cluster/v1alpha1"
	workv1alpha2 "github.com/karmada-io/karmada/pkg/apis/work/v1alpha2"
	estimatorclient "github.com/karmada-io/karmada/pkg/estimator/client"
	"github.com/karmada-io/karmada/pkg/features"
)

func TestK014ProbeG1DefaultResourceModels(t *testing.T) {
	const gpu = corev1.ResourceName("nvidia.com/gpu")
	// 4 nodes, each 32 CPU / 128Gi / 4 GPU, nothing running.
	var nodes []*corev1.Node
	total := corev1.ResourceList{}
	for _, name := range []string{"n1", "n2", "n3", "n4"} {
		alloc := corev1.ResourceList{
			corev1.ResourceCPU:    resource.MustParse("32"),
			corev1.ResourceMemory: resource.MustParse("128Gi"),
			corev1.ResourcePods:   resource.MustParse("110"),
			gpu:                   resource.MustParse("4"),
		}
		nodes = append(nodes, &corev1.Node{ObjectMeta: metav1.ObjectMeta{Name: name}, Status: corev1.NodeStatus{Allocatable: alloc}})
		for k, v := range alloc {
			q := total[k]
			q.Add(v)
			total[k] = q
		}
	}

	// What the aggregated apiserver writes into Cluster.spec.resourceModels on creation.
	internal := &clusterapis.Cluster{}
	mutation.SetDefaultClusterResourceModels(internal)
	cluster := &clusterv1alpha1.Cluster{ObjectMeta: metav1.ObjectMeta{Name: "member"}}
	if err := clusterv1alpha1.Convert_cluster_ClusterSpec_To_v1alpha1_ClusterSpec(&internal.Spec, &cluster.Spec, nil); err != nil {
		t.Fatal(err)
	}
	// What the cluster status controller reports.
	cluster.Status.ResourceSummary = &clusterv1alpha1.ResourceSummary{
		Allocatable:          total,
		AllocatableModelings: getAllocatableModelings(cluster, nodes, nil),
	}
	t.Logf("K014 G1 | gate CustomizedClusterResourceModeling=%v | default models=%d grades, dimensions of grade 0=%v | allocatableModelings=%v",
		features.FeatureGate.Enabled(features.CustomizedClusterResourceModeling), len(cluster.Spec.ResourceModels),
		func() (names []corev1.ResourceName) {
			for _, r := range cluster.Spec.ResourceModels[0].Ranges {
				names = append(names, r.Name)
			}
			return
		}(), cluster.Status.ResourceSummary.AllocatableModelings)

	ge := estimatorclient.NewGeneralEstimator()
	ask := func(label string, c *clusterv1alpha1.Cluster, req corev1.ResourceList) {
		res, err := ge.MaxAvailableReplicas(context.TODO(), estimatorclient.ReplicaEstimationRequest{
			Clusters:            []*clusterv1alpha1.Cluster{c},
			ReplicaRequirements: &workv1alpha2.ReplicaRequirements{ResourceRequest: req},
		})
		t.Logf("K014 G1 | %s | maxAvailableReplicas=%d err=%v", label, res[0].Replicas, err)
	}
	cpuOnly := corev1.ResourceList{corev1.ResourceCPU: resource.MustParse("4"), corev1.ResourceMemory: resource.MustParse("16Gi")}
	oneGPU := corev1.ResourceList{corev1.ResourceCPU: resource.MustParse("4"), corev1.ResourceMemory: resource.MustParse("16Gi"), gpu: resource.MustParse("1")}

	ask("default models, request 4cpu/16Gi (control)", cluster, cpuOnly)
	ask("default models, request 4cpu/16Gi/1gpu (a plain 1-GPU pod)", cluster, oneGPU)

	noModels := cluster.DeepCopy()
	noModels.Spec.ResourceModels = nil
	noModels.Status.ResourceSummary.AllocatableModelings = getAllocatableModelings(noModels, nodes, nil)
	ask("resourceModels cleared, request 4cpu/16Gi/1gpu", noModels, oneGPU)
}
