// Probe (temporary test, not part of any change): what the node-level
// scheduler-estimator answers for a ServingGroup under each requirement model.

import (
	"fmt"
	"testing"

	appsv1 "k8s.io/api/apps/v1"
	corev1 "k8s.io/api/core/v1"
	"k8s.io/apimachinery/pkg/api/resource"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
	"k8s.io/apimachinery/pkg/runtime"
	"k8s.io/apimachinery/pkg/runtime/schema"
	discoveryfake "k8s.io/client-go/discovery/fake"
	dynamicfake "k8s.io/client-go/dynamic/fake"
	"k8s.io/client-go/kubernetes/fake"
	coretesting "k8s.io/client-go/testing"

	"github.com/karmada-io/karmada/cmd/scheduler-estimator/app/options"
	"github.com/karmada-io/karmada/pkg/estimator/pb"
	testhelper "github.com/karmada-io/karmada/test/helper"
)

const k014GPU = corev1.ResourceName("nvidia.com/gpu")

func k014Node(name string, gpus int64) *corev1.Node {
	n := testhelper.NewNode(name, 64000, 512*1024*1024*1024, 110, 1024*1024*1024*1024)
	q := *resource.NewQuantity(gpus, resource.DecimalSI)
	n.Status.Capacity[k014GPU] = q
	n.Status.Allocatable[k014GPU] = q
	return n
}

func k014Req(gpus int64) corev1.ResourceList {
	return corev1.ResourceList{
		corev1.ResourceCPU:    resource.MustParse("1"),
		corev1.ResourceMemory: resource.MustParse("1Gi"),
		k014GPU:               *resource.NewQuantity(gpus, resource.DecimalSI),
	}
}

type k014Pod struct {
	name     string
	gpus     int64
	replicas int32
}

func TestK014ProbeAccurateEstimator(t *testing.T) {
	clusters := map[string][]runtime.Object{
		"frag(2 nodes x 2 GPU)": {k014Node("n1", 2), k014Node("n2", 2)},
		"big(1 node x 8 GPU)":   {k014Node("n1", 8)},
	}
	shapes := map[string][]k014Pod{
		"4x1GPU(entry1+3 workers)": {{"entry", 1, 1}, {"worker", 1, 3}},
		"3GPU+1GPU(entry+1 worker)": {{"entry", 3, 1}, {"worker", 1, 1}},
	}
	for cname, objs := range clusters {
		for sname, pods := range shapes {
			t.Run(cname+"/"+sname, func(t *testing.T) {
				ctx := t.Context()
				gvrToListKind := map[schema.GroupVersionResource]string{
					{Group: "apps", Version: "v1", Resource: "deployments"}: "DeploymentList",
				}
				dynamicClient := dynamicfake.NewSimpleDynamicClientWithCustomListKinds(runtime.NewScheme(), gvrToListKind)
				discoveryClient := &discoveryfake.FakeDiscovery{Fake: &coretesting.Fake{}}
				discoveryClient.Resources = []*metav1.APIResourceList{{
					GroupVersion: appsv1.SchemeGroupVersion.String(),
					APIResources: []metav1.APIResource{{Name: "deployments", Namespaced: true, Kind: "Deployment"}},
				}}
				es, err := NewEstimatorServer(ctx, fake.NewClientset(objs...), dynamicClient, discoveryClient, &options.Options{ClusterName: "fake"})
				if err != nil {
					t.Fatal(err)
				}
				es.informerFactory.Start(ctx.Done())
				es.informerFactory.WaitForCacheSync(ctx.Done())
				es.informerManager.WaitForCacheSync()
				waitForEstimatorCacheSync(t, es, objs)

				var sum, maxPod int64
				var cpuSum, memSum resource.Quantity
				comps := []*pb.Component{}
				for _, p := range pods {
					sum += p.gpus * int64(p.replicas)
					if p.gpus > maxPod {
						maxPod = p.gpus
					}
					for i := int32(0); i < p.replicas; i++ {
						cpuSum.Add(resource.MustParse("1"))
						memSum.Add(resource.MustParse("1Gi"))
					}
					comps = append(comps, &pb.Component{
						Name:                p.name,
						Replicas:            p.replicas,
						ReplicaRequirements: (&pb.ComponentReplicaRequirements{}).MustSetResourceRequest(k014Req(p.gpus)),
					})
				}
				sumReq := corev1.ResourceList{corev1.ResourceCPU: cpuSum, corev1.ResourceMemory: memSum, k014GPU: *resource.NewQuantity(sum, resource.DecimalSI)}

				out := ""
				r1, err := es.MaxAvailableReplicas(ctx, &pb.MaxAvailableReplicasRequest{Cluster: "fake", ReplicaRequirements: (&pb.ReplicaRequirements{}).MustSetResourceRequest(sumReq)})
				out += fmt.Sprintf("replicas(sum=%dGPU)=%v err=%v; ", sum, r1.GetMaxReplicas(), err)
				r2, err := es.MaxAvailableReplicas(ctx, &pb.MaxAvailableReplicasRequest{Cluster: "fake", ReplicaRequirements: (&pb.ReplicaRequirements{}).MustSetResourceRequest(k014Req(maxPod))})
				out += fmt.Sprintf("replicas(maxPod=%dGPU)=%v err=%v; ", maxPod, r2.GetMaxReplicas(), err)
				r3, err := es.MaxAvailableComponentSets(ctx, &pb.MaxAvailableComponentSetsRequest{Cluster: "fake", Namespace: "default", Components: comps})
				out += fmt.Sprintf("componentSets=%v err=%v", r3.GetMaxSets(), err)
				t.Logf("K014 accurate | %s | %s | %s", cname, sname, out)
			})
		}
	}
}
